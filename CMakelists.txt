set(CMAKE_MODULE_PATH ${CMAKE_MODULE_PATH} ${CMAKE_CURRENT_SOURCE_DIR}/cmake)
include("SetupVCPKG")

cmake_minimum_required(VERSION 3.21.0)
project(gsync VERSION 0.1.0)

include("SetupOutputDir")

if(CMAKE_CXX_COMPILER_ID MATCHES MSVC)
  add_compile_options("/permissive-")
endif()

if(CMAKE_SYSTEM_NAME STREQUAL "Windows")
  add_definitions(-DWIN32_LEAN_AND_MEAN -DWIN32 -DNOMINMAX)
  add_compile_options(/EHsc)
endif()

set(CMAKE_CXX_STANDARD 20)

add_compile_definitions(-DBUILDING_GSYNC_SHARED=1)

include_directories("${CMAKE_SOURCE_DIR}/include")

set(BUILD_MQAS_EXAMPLES OFF CACHE BOOL "Compile mqas examples" FORCE)
set(MQAS_BUILD_SHARED_LIBS ON CACHE BOOL "Build shared mqas" FORCE)

set(GSY_SOURCE_FILES
        "${CMAKE_CURRENT_SOURCE_DIR}/src/core/engine.cpp"
        "${CMAKE_CURRENT_SOURCE_DIR}/src/core/connection.cpp"
)


add_subdirectory(mqas)

add_subdirectory(example1)
