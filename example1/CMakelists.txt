
add_custom_command(
        OUTPUT ${CMAKE_CURRENT_SOURCE_DIR}/p2p_datagram.pb.cc ${CMAKE_CURRENT_SOURCE_DIR}/p2p_datagram.pb.h
        COMMAND $<TARGET_FILE:protobuf::protoc> --cpp_out=. p2p_datagram.proto
        DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/p2p_datagram.proto protobuf::protoc
        WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}
        COMMENT "Generating p2p_datagram proto"
)
add_custom_target(gsync_e1_proto DEPENDS
        ${CMAKE_CURRENT_SOURCE_DIR}/p2p_datagram.pb.cc
        ${CMAKE_CURRENT_SOURCE_DIR}/p2p_datagram.pb.h)

add_library(gsync_e1 SHARED
        lib.cpp lobby.cpp
        p2p_datagram_stream.cpp
        p2p_datagram.pb.cc
        ${GSY_SOURCE_FILES})

add_dependencies(gsync_e1 gsync_e1_proto)

add_dependencies(gsync_e1 mqas)

target_link_libraries(gsync_e1 PUBLIC mqas)

add_custom_command(TARGET gsync_e1 POST_BUILD
        COMMAND ${CMAKE_COMMAND}
        -D TARGET_DIR="$<TARGET_FILE_DIR:gsync_e1>"
        -D FILE_LIST="$<TARGET_FILE_DIR:mqas>/dependencies.ini"
        -P "${CMAKE_SOURCE_DIR}/cmake/CopyFiles.cmake"
        COMMENT "Copying mqas dependencies binary files"
)

if(GSYNC_BUILD_UNIT_TEST)
        add_executable(te1 main.cpp)
        target_link_libraries(te1 PRIVATE gsync_e1)
        target_compile_definitions(te1 PRIVATE CONFIG_PATH="${CMAKE_CURRENT_SOURCE_DIR}/conf.txt")
endif()
